import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { Property } from '../models/property.model';

@Injectable({
  providedIn: 'root'
})
export class PropertyService {

  public apiUrl="https://ide-ffbbcacca341114013aababaebabeone.premiumproject.examly.io/proxy/3001/properties";

  constructor(public ser:HttpClient) { }

  getProperties():Observable<Property[]>{
    return this.ser.get<Property[]>(this.apiUrl);
  }

  getPropertyById(id:number):Observable<Property>{
    return this.ser.get<Property>(`${this.apiUrl}/${id}`);
  }

  deletepropertyById(id:number){
    return this.ser.delete(`${this.apiUrl}/${id}`);
  }

  addProperty(obj:Property){
    return this.ser.post<Property>(this.apiUrl,obj)
  }
}

import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { PropertyService } from 'src/app/services/property.service';
import { Property } from 'src/app/models/property.model';

@Component({
  selector: 'app-delete-confirm',
  templateUrl: './delete-confirm.component.html',
  styleUrls: ['./delete-confirm.component.css']
})
export class DeleteConfirmComponent implements OnInit {

  pid:number;
  prop:Property={
    id:0,title:'', location:'', price:0,size:0,type:'',available:false
  }

  constructor(public ser:PropertyService, public ac:ActivatedRoute, public rt:Router) { }

  ngOnInit() {
    this.ac.params.subscribe(r=>{
      this.pid=+r['id'];

      this.ser.getPropertyById(this.pid).subscribe(res=>this.prop=res)
    })
  }

  deleteProperty(id:number){
    this.ser.deletepropertyById(id).subscribe(r=>this.rt.navigate([`/getProperties`]))
  }

  cancelProperty(){
    this.rt.navigate([`/getProperties`]);
  }


}

import { Component, OnInit } from '@angular/core';
import { PropertyService } from 'src/app/services/property.service';
import { Router } from '@angular/router';
import { Property } from 'src/app/models/property.model';

@Component({
  selector: 'app-add-property',
  templateUrl: './add-property.component.html',
  styleUrls: ['./add-property.component.css']
})
export class AddPropertyComponent implements OnInit {
  pid:number;

  prop:Property={
    id:0,title:'', location:'', price:0,size:0,type:'',available:false
  }
  formsubmitted:boolean=false
  constructor(public ser:PropertyService,public rt:Router) { }

  ngOnInit():void {
  }

  addProperty(){
    this.formsubmitted=true;
    if(this.prop.id && this.prop.title && this.prop.available && this.prop.price){
      this.ser.addProperty(this.prop).subscribe(r=>this.rt.navigate([`/getProperties`]));
    }
  }

}
